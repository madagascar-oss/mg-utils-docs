import { Card, CardGrid, Tabs, TabItem } from '@astrojs/starlight/components';

<Tabs syncKey="language">
  <TabItem label="JavaScript">
    ```sh title="Terminal"
    npm install @madagascar-oss/utils
    ```
    ```ts
    import { formatAriary, formatPhoneNumber, numberToWords } from '@madagascar-oss/utils';

    numberToWords(2026); // 'enina amby roapolo sy roa arivo'
    formatPhoneNumber('0341234567', 'international'); // '+261 34 12 345 67'
    formatAriary(1500); // 'Ar 1 500'
    ```
  </TabItem>
  <TabItem label="PHP">
    ```sh title="Terminal"
    composer require madagascar-oss/mg-utils
    ```
    ```php
    use Madagascar\MgUtils\Ariary\Ariary;
    use Madagascar\MgUtils\Phone\PhoneNumber;
    use Madagascar\MgUtils\Phone\PhoneNumberFormat;
    use Madagascar\MgUtils\Words\NumberWords;

    NumberWords::fromInt(2026); // 'enina amby roapolo sy roa arivo'
    PhoneNumber::parse('0341234567')->format(PhoneNumberFormat::International); // '+261 34 12 345 67'
    Ariary::fromInt(1500)->format(); // 'Ar 1 500'
    ```
  </TabItem>
</Tabs>
