<Tabs syncKey="language">
  <TabItem label="JavaScript">
    ```sh title="Terminal"
    npm install @mdgascar/utils
    ```
    ```ts
    import { formatAriary, formatPhoneNumber, numberToWords } from '@mdgascar/utils';

    numberToWords(2026); // 'enina amby roapolo sy roa arivo'
    formatPhoneNumber('0341234567', 'international'); // '+261 34 12 345 67'
    formatAriary(1500); // 'Ar 1 500'
    ```
  </TabItem>
  <TabItem label="PHP">
    ```sh title="Terminal"
    composer require mdgascar/mdg-utils
    ```
    ```php
    use Mdgascar\MdgUtils\Ariary\Ariary;
    use Mdgascar\MdgUtils\Phone\PhoneNumber;
    use Mdgascar\MdgUtils\Phone\PhoneNumberFormat;
    use Mdgascar\MdgUtils\Words\NumberWords;

    NumberWords::fromInt(2026); // 'enina amby roapolo sy roa arivo'
    PhoneNumber::parse('0341234567')->format(PhoneNumberFormat::International); // '+261 34 12 345 67'
    Ariary::fromInt(1500)->format(); // 'Ar 1 500'
    ```
  </TabItem>
</Tabs>
